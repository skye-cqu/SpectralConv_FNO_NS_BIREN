1. 赛题简介
Fourier Neural Operator（FNO）是 AI for Science 领域最具代表性的神经算子架构之一，广泛应用于流体力学模拟、天气预报、地下水流预测等偏微分方程（PDE）求解场景。FNO 的核心创新在于 Spectral Convolution（谱卷积）算子——在傅里叶频域空间中学习全局卷积核，相比传统卷积具有分辨率无关性和全局感受野等优势。
Spectral Convolution 的计算流程为：
输入特征 x(shape: [B, C_in, H, W])
→ FFT 变换到频域
→ 频域复数矩阵乘法（可学习权重 R，shape: [C_in, C_out, modes, modes]）
→ IFFT 变换回空间域
→ 输出特征 y(shape: [B, C_out, H, W])
本题要求参赛者在壁仞 GPU 平台上实现高性能的 Spectral Convolution 算子，并通过正确性验证和性能测试。注意：FFT 运算建议优先支持空间维度为 2 的幂次（如 64, 128, 256）的输入 shape，以获得最佳计算性能。

python复制import torch
import torch.nn as nn
import torch.fft
import time
import numpy as np

class SpectralConv2d(nn.Module):
    """2D Spectral Convolution 算子 - FNO核心组件

    计算流程: FFT -> 频域复数矩阵乘(截断低频) -> IFFT
    """
    def __init__(self, in_channels, out_channels, modes1, modes2):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.modes1 = modes1  # 第一维频率截断模态数
        self.modes2 = modes2  # 第二维频率截断模态数

        scale = 1.0 / (in_channels * out_channels)
        # 可学习的频域权重 (两组: 正频率区域 + 负频率区域)
        self.weights1 = nn.Parameter(
            scale * torch.rand(in_channels, out_channels, modes1, modes2, dtype=torch.cfloat))
        self.weights2 = nn.Parameter(
            scale * torch.rand(in_channels, out_channels, modes1, modes2, dtype=torch.cfloat))

    def compl_mul2d(self, input, weights):
        """复数批量矩阵乘: (B, C_in, H, W) x (C_in, C_out, H, W) -> (B, C_out, H, W)"""
        return torch.einsum("bixy,ioxy->boxy", input, weights)

    def forward(self, x):
        """
        前向计算
        Args:
            x: [B, C_in, H, W] 实数张量
        Returns:
            [B, C_out, H, W] 实数张量
        """
        B, C_in, H, W = x.shape

        # Step 1: 2D实数FFT (rfft2), 输出shape: [B, C_in, H, W//2+1] (复数)
        x_ft = torch.fft.rfft2(x)

        # Step 2: 在频域进行低频模态的复数矩阵乘
        out_ft = torch.zeros(B, self.out_channels, H, W // 2 + 1,
                            dtype=torch.cfloat, device=x.device)

        # 正频率部分 [:modes1, :modes2]
        out_ft[:, :, :self.modes1, :self.modes2] = \
            self.compl_mul2d(x_ft[:, :, :self.modes1, :self.modes2], self.weights1)

        # 负频率部分 [-modes1:, :modes2]
        out_ft[:, :, -self.modes1:, :self.modes2] = \
            self.compl_mul2d(x_ft[:, :, -self.modes1:, :self.modes2], self.weights2)

        # Step 3: 逆FFT回空间域
        x_out = torch.fft.irfft2(out_ft, s=(H, W))
        return x_out


class SpectralConv3d(nn.Module):
    """3D Spectral Convolution 算子 (进阶)"""
    def __init__(self, in_channels, out_channels, modes1, modes2, modes3):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.modes1 = modes1
        self.modes2 = modes2
        self.modes3 = modes3

        scale = 1.0 / (in_channels * out_channels)
        self.weights1 = nn.Parameter(scale * torch.rand(
            in_channels, out_channels, modes1, modes2, modes3, dtype=torch.cfloat))
        self.weights2 = nn.Parameter(scale * torch.rand(
            in_channels, out_channels, modes1, modes2, modes3, dtype=torch.cfloat))
        self.weights3 = nn.Parameter(scale * torch.rand(
            in_channels, out_channels, modes1, modes2, modes3, dtype=torch.cfloat))
        self.weights4 = nn.Parameter(scale * torch.rand(
            in_channels, out_channels, modes1, modes2, modes3, dtype=torch.cfloat))

    def compl_mul3d(self, input, weights):
        return torch.einsum("bixyz,ioxyz->boxyz", input, weights)

    def forward(self, x):
        B, C, D, H, W = x.shape
        x_ft = torch.fft.rfftn(x, dim=[-3, -2, -1])

        out_ft = torch.zeros(B, self.out_channels, D, H, W // 2 + 1,
                            dtype=torch.cfloat, device=x.device)

        out_ft[:, :, :self.modes1, :self.modes2, :self.modes3] = \
            self.compl_mul3d(x_ft[:, :, :self.modes1, :self.modes2, :self.modes3], self.weights1)
        out_ft[:, :, -self.modes1:, :self.modes2, :self.modes3] = \
            self.compl_mul3d(x_ft[:, :, -self.modes1:, :self.modes2, :self.modes3], self.weights2)
        out_ft[:, :, :self.modes1, -self.modes2:, :self.modes3] = \
            self.compl_mul3d(x_ft[:, :, :self.modes1, -self.modes2:, :self.modes3], self.weights3)
        out_ft[:, :, -self.modes1:, -self.modes2:, :self.modes3] = \
            self.compl_mul3d(x_ft[:, :, -self.modes1:, -self.modes2:, :self.modes3], self.weights4)

        x_out = torch.fft.irfftn(out_ft, s=(D, H, W))
        return x_out

        # 前向计时
        start = time.time()
        for _ in range(100):
            _ = model(x)
        torch.cuda.synchronize()

            "resolution": f"{H}x{W}",
            "forward_time_ms": f"{fwd_time:.3f}",
            "memory_MB": f"{mem:.1f}",
        })
        print(f"分辨率 {H}x{W}: 前向 {fwd_time:.3f}ms, 显存 {mem:.1f}MB")

    return results

1. 赛题简介
Physics-Informed Neural Network（PINN）是将物理定律嵌入神经网络训练过程的经典方法，通过将 PDE 残差作为损失函数的一部分，实现无需标注数据即可求解微分方程。
求解目标: 二维稳态热传导方程（Laplace/Poisson 方程）:
∂²u/∂x² + ∂²u/∂y² = f(x, y), (x, y) ∈ Ω
应用场景：芯片散热分析、建筑热工设计、地热资源评估。

python复制import torch
import torch.nn as nn
import numpy as np
import matplotlib.pyplot as plt

class PINN(nn.Module):
    """Physics-Informed Neural Network for 2D heat equation"""
    def __init__(self, layers=[2, 64, 64, 64, 64, 1], activation=nn.Tanh):
        super().__init__()
        modules = []
        for i in range(len(layers) - 1):
            modules.append(nn.Linear(layers[i], layers[i+1]))
            if i < len(layers) - 2:
                modules.append(activation())
        self.network = nn.Sequential(*modules)
        self._initialize_weights()

    def _initialize_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Linear):
                nn.init.xavier_normal_(m.weight)
                nn.init.zeros_(m.bias)

    def forward(self, x):
        """x: [N, 2] -> u: [N, 1]"""
        return self.network(x)


class PINNSolver:
    """PINN求解器: 求解 Δu = f(x,y) on [0,1]x[0,1]"""

    def __init__(self, device='cuda'):
        self.device = torch.device(device if torch.cuda.is_available() else 'cpu')
        self.model = PINN(layers=[2, 128, 128, 128, 128, 1]).to(self.device)
        self.optimizer = torch.optim.Adam(self.model.parameters(), lr=1e-3)
        self.scheduler = torch.optim.lr_scheduler.StepLR(self.optimizer, step_size=2000, gamma=0.5)

    def exact_solution(self, x, y):
        """解析解: u(x,y) = sin(πx)·sin(πy)"""
        return torch.sin(np.pi * x) * torch.sin(np.pi * y)

    def source_term(self, x, y):
        """源项: f(x,y) = -2π²·sin(πx)·sin(πy)"""
        return -2 * np.pi**2 * torch.sin(np.pi * x) * torch.sin(np.pi * y)

    def compute_pde_residual(self, xy):
        """计算PDE残差: Δu - f = 0"""
        xy.requires_grad_(True)
        u = self.model(xy)

        # 一阶导数
        grad_u = torch.autograd.grad(u, xy, grad_outputs=torch.ones_like(u),
                                      create_graph=True)[0]
        u_x = grad_u[:, 0:1]
        u_y = grad_u[:, 1:2]

        # 二阶导数
        u_xx = torch.autograd.grad(u_x, xy, grad_outputs=torch.ones_like(u_x),
                                    create_graph=True)[0][:, 0:1]
        u_yy = torch.autograd.grad(u_y, xy, grad_outputs=torch.ones_like(u_y),
                                    create_graph=True)[0][:, 1:2]

        # PDE残差: Δu - f
        f = self.source_term(xy[:, 0:1], xy[:, 1:2])
        residual = u_xx + u_yy - f
        return residual

    def compute_bc_loss(self, n_bc=200):
        """计算边界条件损失: u = 0 on ∂Ω"""
        # 四条边界
        t = torch.linspace(0, 1, n_bc, device=self.device).unsqueeze(1)
        zeros = torch.zeros_like(t)
        ones = torch.ones_like(t)

        bc_points = torch.cat([
            torch.cat([t, zeros], dim=1),     # 下边 y=0
            torch.cat([t, ones], dim=1),      # 上边 y=1
            torch.cat([zeros, t], dim=1),     # 左边 x=0
            torch.cat([ones, t], dim=1),      # 右边 x=1
        ], dim=0)

        u_bc = self.model(bc_points)
        return torch.mean(u_bc**2)

    def train(self, n_epochs=10000, n_interior=2000, n_bc=200):
        """训练PINN"""
        losses = []
        for epoch in range(n_epochs):
            self.optimizer.zero_grad()

            # 内部采样点
            xy_interior = torch.rand(n_interior, 2, device=self.device)

            # PDE残差损失
            residual = self.compute_pde_residual(xy_interior)
            loss_pde = torch.mean(residual**2)

            # 边界条件损失
            loss_bc = self.compute_bc_loss(n_bc)

            # 总损失
            loss = loss_pde + 100.0 * loss_bc  # 边界条件权重更大

            loss.backward()
            self.optimizer.step()
            self.scheduler.step()

            losses.append(loss.item())

            if (epoch + 1) % 1000 == 0:
                l2_err = self.compute_l2_error()
                print(f"Epoch {epoch+1}: Loss={loss.item():.6f}, "
                      f"PDE={loss_pde.item():.6f}, BC={loss_bc.item():.6f}, "
                      f"L2 Error={l2_err:.6f}")

        return losses

    def compute_l2_error(self, n=100):
        """计算相对L2误差"""
        x = torch.linspace(0, 1, n, device=self.device)
        y = torch.linspace(0, 1, n, device=self.device)
        X, Y = torch.meshgrid(x, y, indexing='ij')
        xy = torch.stack([X.flatten(), Y.flatten()], dim=1)

        with torch.no_grad():
            u_pred = self.model(xy).reshape(n, n)
        u_exact = self.exact_solution(X, Y)

        l2_error = torch.norm(u_pred - u_exact) / torch.norm(u_exact)
        return l2_error.item()

    def visualize(self, save_path='pinn_result.png'):
        """可视化结果"""
        n = 100
        x = torch.linspace(0, 1, n, device=self.device)
        y = torch.linspace(0, 1, n, device=self.device)
        X, Y = torch.meshgrid(x, y, indexing='ij')
        xy = torch.stack([X.flatten(), Y.flatten()], dim=1)

        with torch.no_grad():
            u_pred = self.model(xy).reshape(n, n).cpu().numpy()
        u_exact = self.exact_solution(X, Y).cpu().numpy()

        fig, axes = plt.subplots(1, 3, figsize=(15, 4))
        im0 = axes[0].imshow(u_exact, extent=[0,1,0,1], origin='lower', cmap='hot')
        axes[0].set_title('Exact Solution')
        plt.colorbar(im0, ax=axes[0])

        im1 = axes[1].imshow(u_pred, extent=[0,1,0,1], origin='lower', cmap='hot')
        axes[1].set_title('PINN Prediction')
        plt.colorbar(im1, ax=axes[1])

        im2 = axes[2].imshow(np.abs(u_pred - u_exact), extent=[0,1,0,1], origin='lower', cmap='hot')
        axes[2].set_title('Absolute Error')
        plt.colorbar(im2, ax=axes[2])

        plt.tight_layout()
        plt.savefig(save_path, dpi=150)
        print(f"结果已保存到 {save_path}")

    solver.visualize()

import torch.nn as nn
import torch.nn.functional as F
from torch.utils.data import Dataset, DataLoader
import numpy as np

class EdgeModel(nn.Module):
    """边更新模型"""
    def __init__(self, node_dim, edge_dim, hidden_dim):
        super().__init__()
        self.edge_mlp = nn.Sequential(
            nn.Linear(2 * node_dim + edge_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, edge_dim)
        )

    def forward(self, src, dst, edge_attr):
        """src: [E, node_dim], dst: [E, node_dim], edge_attr: [E, edge_dim]"""
        out = torch.cat([src, dst, edge_attr], dim=-1)
        return self.edge_mlp(out)


class NodeModel(nn.Module):
    """节点更新模型"""
    def __init__(self, node_dim, edge_dim, hidden_dim):
        super().__init__()
        self.node_mlp = nn.Sequential(
            nn.Linear(node_dim + edge_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, node_dim)
        )

    def forward(self, x, edge_index, edge_attr):
        """
        x: [N, node_dim], edge_index: [2, E], edge_attr: [E, edge_dim]
        """
        row, col = edge_index  # row->col 消息传递方向
        # 聚合邻居边信息
        agg = torch.zeros_like(x[:, :edge_attr.shape[1]])
        # scatter_add 实现
        agg = torch.zeros(x.shape[0], edge_attr.shape[1], device=x.device)
        agg.scatter_add_(0, col.unsqueeze(1).expand_as(edge_attr), edge_attr)

        out = torch.cat([x, agg], dim=-1)
        return self.node_mlp(out)


class MPNN(nn.Module):
    """Message Passing Neural Network for molecular property prediction"""
    def __init__(self, node_input_dim=11, edge_input_dim=4, hidden_dim=128,
                 output_dim=1, n_layers=3):
        super().__init__()
        self.node_embed = nn.Linear(node_input_dim, hidden_dim)
        self.edge_embed = nn.Linear(edge_input_dim, hidden_dim)

        self.message_layers = nn.ModuleList()
        self.update_layers = nn.ModuleList()
        for _ in range(n_layers):
            self.message_layers.append(EdgeModel(hidden_dim, hidden_dim, hidden_dim))
            self.update_layers.append(NodeModel(hidden_dim, hidden_dim, hidden_dim))

        # 全局读出 (readout)
        self.readout = nn.Sequential(
            nn.Linear(hidden_dim, hidden_dim),
            nn.ReLU(),
            nn.Linear(hidden_dim, hidden_dim // 2),
            nn.ReLU(),
            nn.Linear(hidden_dim // 2, output_dim)
        )

    def forward(self, x, edge_index, edge_attr, batch):
        """
        Args:
            x: [N_total, node_input_dim] 节点特征
            edge_index: [2, E_total] 边索引
            edge_attr: [E_total, edge_input_dim] 边特征
            batch: [N_total] 每个节点属于哪个图
        Returns:
            [B, output_dim] 每个图的预测
        """
        # 嵌入
        h = self.node_embed(x)
        e = self.edge_embed(edge_attr)

        # 消息传递
        for msg_layer, upd_layer in zip(self.message_layers, self.update_layers):
            row, col = edge_index
            # 边更新
            e = e + msg_layer(h[row], h[col], e)
            # 节点更新
            h = h + upd_layer(h, edge_index, e)

        # 图级别聚合 (sum pooling)
        num_graphs = batch.max().item() + 1
        graph_repr = torch.zeros(num_graphs, h.shape[1], device=h.device)
        graph_repr.scatter_add_(0, batch.unsqueeze(1).expand_as(h), h)

        # 预测
        out = self.readout(graph_repr)
        return out


class SimpleMolDataset(Dataset):
    """简化版分子数据集 (演示用, 实际使用QM9)"""
    def __init__(self, n_molecules=1000, max_atoms=9):

    def _generate_dummy_data(self):

        for _ in range(self.n_molecules):
            n_atoms = np.random.randint(3, self.max_atoms + 1)
            x = torch.randn(n_atoms, 11)  # 原子特征 (原子类型one-hot等)
            # 随机连接

                    edges.append([i, j])
                    edges.append([j, i])
            edge_index = torch.tensor(edges, dtype=torch.long).t()
            edge_attr = torch.randn(edge_index.shape[1], 4)  # 边特征

def collate_graphs(batch):
    """将多个图合并为一个大图 (batching)"""
    xs, edge_indices, edge_attrs, ys = [], [], [], []
    batch_idx = []
    node_offset = 0

    for i, (x, edge_index, edge_attr, y) in enumerate(batch):
        xs.append(x)
        edge_indices.append(edge_index + node_offset)
        edge_attrs.append(edge_attr)
        ys.append(y)
        batch_idx.append(torch.full((x.shape[0],), i, dtype=torch.long))
        node_offset += x.shape[0]

    return (torch.cat(xs, 0), torch.cat(edge_indices, 1),
            torch.cat(edge_attrs, 0), torch.cat(ys, 0),
            torch.cat(batch_idx, 0))


def train_gnn():
    """训练GNN模型"""
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

    # 数据
    dataset = SimpleMolDataset(n_molecules=5000)
    train_size = int(0.8 * len(dataset))
    train_set, val_set = torch.utils.data.random_split(
        dataset, [train_size, len(dataset) - train_size])

    train_loader = DataLoader(train_set, batch_size=32, shuffle=True, collate_fn=collate_graphs)
    val_loader = DataLoader(val_set, batch_size=32, collate_fn=collate_graphs)

    # 模型
    model = MPNN(node_input_dim=11, edge_input_dim=4, hidden_dim=128, output_dim=1).to(device)
    optimizer = torch.optim.Adam(model.parameters(), lr=1e-3)
    scheduler = torch.optim.lr_scheduler.ReduceLROnPlateau(optimizer, patience=10, factor=0.5)

    # 训练
    for epoch in range(100):
        model.train()
        total_loss = 0
        for x, edge_index, edge_attr, y, batch_idx in train_loader:
            x, edge_index, edge_attr = x.to(device), edge_index.to(device), edge_attr.to(device)
            y, batch_idx = y.to(device), batch_idx.to(device)

            optimizer.zero_grad()
            pred = model(x, edge_index, edge_attr, batch_idx)
            loss = F.mse_loss(pred.squeeze(), y)
            loss.backward()
            optimizer.step()
            total_loss += loss.item()

        # 验证
        model.eval()
        val_mae = 0
        n_val = 0
        with torch.no_grad():
            for x, edge_index, edge_attr, y, batch_idx in val_loader:
                x, edge_index, edge_attr = x.to(device), edge_index.to(device), edge_attr.to(device)
                y, batch_idx = y.to(device), batch_idx.to(device)
                pred = model(x, edge_index, edge_attr, batch_idx)
                val_mae += torch.abs(pred.squeeze() - y).sum().item()
                n_val += y.shape[0]

        val_mae /= n_val
        scheduler.step(val_mae)

        if (epoch + 1) % 10 == 0:
            print(f"Epoch {epoch+1}: Train Loss={total_loss/len(train_loader):.6f}, Val MAE={val_mae:.6f}")

    return model


if __name__ == "__main__":
    model = train_gnn()
    print("训练完成!")

python复制import torch
import torch.nn as nn
import torch.nn.functional as F
import numpy as np
import h5py
from torch.utils.data import Dataset, DataLoader

class SpectralConv2d(nn.Module):
    """2D Spectral Convolution (复用必选题实现)"""
    def __init__(self, in_channels, out_channels, modes1, modes2):
        super().__init__()
        self.in_channels = in_channels
        self.out_channels = out_channels
        self.modes1 = modes1
        self.modes2 = modes2
        scale = 1.0 / (in_channels * out_channels)
        self.weights1 = nn.Parameter(
            scale * torch.rand(in_channels, out_channels, modes1, modes2, dtype=torch.cfloat))
        self.weights2 = nn.Parameter(
            scale * torch.rand(in_channels, out_channels, modes1, modes2, dtype=torch.cfloat))

    def compl_mul2d(self, input, weights):
        return torch.einsum("bixy,ioxy->boxy", input, weights)

    def forward(self, x):
        B, C, H, W = x.shape
        x_ft = torch.fft.rfft2(x)
        out_ft = torch.zeros(B, self.out_channels, H, W // 2 + 1,
                            dtype=torch.cfloat, device=x.device)
        out_ft[:, :, :self.modes1, :self.modes2] = \
            self.compl_mul2d(x_ft[:, :, :self.modes1, :self.modes2], self.weights1)
        out_ft[:, :, -self.modes1:, :self.modes2] = \
            self.compl_mul2d(x_ft[:, :, -self.modes1:, :self.modes2], self.weights2)
        return torch.fft.irfft2(out_ft, s=(H, W))


class FourierLayer(nn.Module):
    """单个Fourier Layer: spectral conv + skip connection + activation"""
    def __init__(self, width, modes1, modes2):
        super().__init__()
        self.spectral_conv = SpectralConv2d(width, width, modes1, modes2)
        self.conv = nn.Conv2d(width, width, 1)  # 1x1卷积作为skip
        self.norm = nn.InstanceNorm2d(width)

    def forward(self, x):
        return F.gelu(self.norm(self.spectral_conv(x) + self.conv(x)))


class FNO2d(nn.Module):
    """Fourier Neural Operator for 2D Navier-Stokes"""
    def __init__(self, modes1=12, modes2=12, width=32, n_layers=4,
                 in_channels=12, out_channels=1):
        super().__init__()
        self.modes1 = modes1
        self.modes2 = modes2
        self.width = width

        # 输入提升
        self.lift = nn.Conv2d(in_channels + 2, width, 1)  # +2 for grid coordinates

        # Fourier layers
        self.fourier_layers = nn.ModuleList([
            FourierLayer(width, modes1, modes2) for _ in range(n_layers)
        ])

        # 输出投影
        self.project = nn.Sequential(
            nn.Conv2d(width, 128, 1),
            nn.GELU(),
            nn.Conv2d(128, out_channels, 1)
        )

    def get_grid(self, shape, device):
        """生成网格坐标"""
        B, C, H, W = shape
        gridx = torch.linspace(0, 1, H, device=device).reshape(1, 1, H, 1).expand(B, 1, H, W)
        gridy = torch.linspace(0, 1, W, device=device).reshape(1, 1, 1, W).expand(B, 1, H, W)
        return torch.cat([gridx, gridy], dim=1)

    def forward(self, x):
        """
        Args:
            x: [B, T_in, H, W] 输入时间步的涡度场
        Returns:
            [B, 1, H, W] 预测的涡度场
        """
        # 拼接网格坐标
        grid = self.get_grid(x.shape, x.device)
        x = torch.cat([x, grid], dim=1)

        # 提升维度
        x = self.lift(x)

        # Fourier layers
        for layer in self.fourier_layers:
            x = layer(x)

        # 投影到输出
        x = self.project(x)
        return x


class NavierStokesDataset(Dataset):
    """Navier-Stokes 2D 数据集"""
    def __init__(self, data_path, n_train=1000, T_in=10, T_out=1, resolution=64):
        """
        data_path: HDF5文件路径
        数据格式: [N, T, H, W] N个样本, T个时间步, H×W空间分辨率
        """
        self.T_in = T_in
        self.T_out = T_out
        self.resolution = resolution

        # 加载数据 (实际使用时从文件加载)
        # 此处用随机数据演示
        self.data = torch.randn(n_train, T_in + T_out, resolution, resolution)

    def __len__(self):
        return self.data.shape[0]

    def __getitem__(self, idx):
        x = self.data[idx, :self.T_in]   # [T_in, H, W]
        y = self.data[idx, -self.T_out:]  # [T_out, H, W]
        return x, y


def train_fno():
    """训练FNO模型"""
    device = torch.device('cuda' if torch.cuda.is_available() else 'cpu')

    # 超参数
    modes = 12
    width = 32
    n_layers = 4
    T_in = 10
    batch_size = 16
    learning_rate = 1e-3
    n_epochs = 100
    resolution = 64

    # 数据
    train_dataset = NavierStokesDataset(
        data_path=None, n_train=1000, T_in=T_in, resolution=resolution)
    train_loader = DataLoader(train_dataset, batch_size=batch_size, shuffle=True)

    # 模型
    model = FNO2d(modes1=modes, modes2=modes, width=width,
                  n_layers=n_layers, in_channels=T_in, out_channels=1).to(device)

    optimizer = torch.optim.Adam(model.parameters(), lr=learning_rate, weight_decay=1e-4)
    scheduler = torch.optim.lr_scheduler.CosineAnnealingLR(optimizer, T_max=n_epochs)

    print(f"模型参数量: {sum(p.numel() for p in model.parameters()):,}")

    # 训练循环
    for epoch in range(n_epochs):
        model.train()
        total_loss = 0
        for x, y in train_loader:
            x, y = x.to(device), y.to(device)

            optimizer.zero_grad()
            pred = model(x)

            # 相对L2损失
            loss = torch.norm(pred - y) / torch.norm(y)
            loss.backward()
            optimizer.step()
            total_loss += loss.item()

        scheduler.step()

        if (epoch + 1) % 10 == 0:
            avg_loss = total_loss / len(train_loader)
            print(f"Epoch {epoch+1}/{n_epochs}: Relative L2 Loss = {avg_loss:.6f}")

    return model


def evaluate_fno(model, test_loader, device):
    """评估FNO模型"""
    model.eval()
    total_l2 = 0
    n_samples = 0

    with torch.no_grad():
        for x, y in test_loader:
            x, y = x.to(device), y.to(device)
            pred = model(x)
            l2 = torch.norm(pred - y, dim=(-2, -1)) / torch.norm(y, dim=(-2, -1))
            total_l2 += l2.sum().item()
            n_samples += x.shape[0]

    return total_l2 / n_samples


if __name__ == "__main__":
    model = train_fno()
    print("FNO训练完成!")
